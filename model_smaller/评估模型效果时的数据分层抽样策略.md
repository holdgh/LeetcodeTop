### 一、原代码逻辑的解释（为什么循环不会有问题）
`StratifiedShuffleSplit(n_splits=1, ...)`的核心特性：
- `n_splits=1` 表示只生成**1组**训练/测试索引（而非多组）；
- 循环`for train_idx, test_idx in sss.split(...)` 本质上只执行**1次**，因此`sample_df`只会被赋值1次，不存在“取最后一次值”的问题；
- 这种写法是`sklearn`官方推荐的（兼容`n_splits>1`的场景），但对于`n_splits=1`的单抽样场景，确实容易让人误解。

### 二、优化后的无歧义版本（直观且避免误解）
将循环写法改为**直接提取索引**，消除“未初始化+循环赋值”的歧义，同时保留分层抽样的核心逻辑：

模型评估函数（优化分层抽样逻辑，消除变量歧义）
"""

import time
import os
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score
from sklearn.model_selection import StratifiedShuffleSplit
from transformers import AutoModelForSequenceClassification, AutoTokenizer

# ====================== 基础配置 ======================
DEVICE = torch.device("cpu") if not torch.cuda.is_available() else torch.device("cuda:0")
VAL_DATA_PATH = "cleaned_dataset.csv"  # 验证数据集路径

# ====================== 核心工具函数：准确计算模型大小 ======================
def get_accurate_model_size(model_dir):
    """兼容safetensors/bin/pt的模型大小计算"""
    if not os.path.exists(model_dir):
        print(f"⚠️  模型路径不存在：{model_dir}")
        return 0.0
    
    weight_suffixes = [".safetensors", ".bin", ".pt", ".pth", ".ckpt"]
    total_size = 0
    for root, dirs, files in os.walk(model_dir):
        for file in files:
            if any(file.endswith(suffix) for suffix in weight_suffixes):
                file_path = os.path.join(root, file)
                total_size += os.path.getsize(file_path)
    
    return round(total_size / (1024 * 1024), 2)

# ====================== 优化：分层随机抽样函数（无歧义） ======================
def stratified_sample_data(data_path, sample_num=1000, random_state=42):
    """
    分层随机抽样，保证评估集标签分布与原始数据一致
    优化点：消除循环赋值歧义，变量初始化更清晰
    """
    # 1. 加载原始数据并过滤无效值
    df = pd.read_csv(data_path, encoding="utf-8")
    df = df[df["text"].notna() & (df["text"].str.len() > 5)].reset_index(drop=True)
    total_data_num = len(df)
    
    # 2. 检查抽样数量合理性
    if sample_num > total_data_num:
        print(f"⚠️  抽样数量({sample_num})超过数据总量({total_data_num})，使用全部数据")
        sample_df = df.copy()
    else:
        # 3. 初始化分层抽样器（仅生成1组索引）
        sss = StratifiedShuffleSplit(
            n_splits=1,                # 仅1组抽样结果
            test_size=sample_num,      # 直接指定抽样数量（替代比例）
            random_state=random_state  # 固定随机种子
        )
        # 4. 直接提取抽样索引（无循环，消除歧义）
        # split返回迭代器，转为列表后取第0组的测试索引
        test_indices = list(sss.split(df, df["label"]))[0][1]
        # 5. 根据索引抽取样本（显式初始化sample_df）
        sample_df = df.iloc[test_indices].reset_index(drop=True)
    
    # 6. 验证样本分布一致性
    original_dist = df["label"].value_counts(normalize=True).round(4)
    sample_dist = sample_df["label"].value_counts(normalize=True).round(4)
    print(f"\n📊 样本分布验证：")
    print(f"原始数据分布：{original_dist.to_dict()}")
    print(f"抽样数据分布：{sample_dist.to_dict()}")
    
    # 7. 返回抽样结果
    return sample_df["text"].tolist(), sample_df["label"].tolist()

# ====================== 完整评估函数 ======================
def evaluate_model(model_path, data_path, sample_num=1000, random_state=42):
    """
    评估模型效果（分层抽样+CPU/GPU通用+准确计算模型大小）
    """
    # 1. 加载模型和Tokenizer
    tokenizer = AutoTokenizer.from_pretrained(model_path)
    try:
        model = AutoModelForSequenceClassification.from_pretrained(
            model_path,
            num_labels=2,
            device_map={"": DEVICE},
            torch_dtype=torch.float16 if DEVICE.type == "cuda" else torch.float32,
            low_cpu_mem_usage=True,
            safe_serialization=True
        )
    except Exception as e:
        print(f"⚠️  模型加载警告：{e}，尝试降级加载")
        model = AutoModelForSequenceClassification.from_pretrained(
            model_path,
            num_labels=2,
            device_map={"": DEVICE},
            low_cpu_mem_usage=True
        )
    model.eval()

    # 2. 分层随机抽样（优化后无歧义版本）
    texts, labels = stratified_sample_data(data_path, sample_num, random_state)
    actual_sample_num = len(texts)

    # 3. 推理速度测试
    preds = []
    start_total = time.time()
    
    with torch.no_grad():
        for text in texts:
            inputs = tokenizer(
                f"判断情感倾向：{text}",
                max_length=128,
                truncation=True,
                padding="max_length",
                return_tensors="pt"
            ).to(DEVICE)
            
            if DEVICE.type == "cuda":
                torch.cuda.synchronize()
                start = torch.cuda.Event(enable_timing=True)
                end = torch.cuda.Event(enable_timing=True)
                start.record()
            
            outputs = model(**inputs)
            pred = torch.argmax(outputs.logits, dim=-1).cpu().item()
            preds.append(pred)
            
            if DEVICE.type == "cuda":
                end.record()
                torch.cuda.synchronize()

    # 4. 计算核心指标
    total_time = time.time() - start_total
    avg_time_per_sample = total_time / actual_sample_num if actual_sample_num > 0 else 0
    accuracy = accuracy_score(labels, preds) if actual_sample_num > 0 else 0
    f1 = f1_score(labels, preds, average="weighted") if actual_sample_num > 0 else 0
    model_size = get_accurate_model_size(model_path)

    # 5. 整理并打印结果
    result = {
        "模型路径": model_path,
        "模型大小(MB)": model_size,
        "指定抽样数": sample_num,
        "实际评估数": actual_sample_num,
        "总推理时间(s)": round(total_time, 2),
        "单样本平均推理时间(ms)": round(avg_time_per_sample * 1000, 2),
        "准确率": round(accuracy, 4),
        "F1值": round(f1, 4),
        "运行设备": DEVICE.type
    }

    print("\n" + "="*60)
    print(f"模型评估结果（分层抽样）- {os.path.basename(model_path)}")
    print("="*60)
    for k, v in result.items():
        print(f"{k}: {v}")

    return result

## 豆包分析

`n_splits`是`StratifiedShuffleSplit`（分层随机抽样）中一个核心参数，理解它的意义能帮你灵活应对不同的评估场景。我会从**定义、核心作用、使用场景**三个维度讲清楚，结合你之前的评估需求给出具体建议。

### 一、`n_splits`的核心定义
`n_splits`（全称：number of splits）表示：**分层抽样器要生成多少组“训练集/测试集”的索引对**。
- 它的默认值是`10`（sklearn官方默认）；
- 你之前的代码中设置`n_splits=1`，表示只生成**1组**抽样索引（对应1批评估样本）。

### 二、`n_splits`的核心作用（结合例子理解）
我们用你的情感分类数据（总样本1万条，正负样本各50%）举例：

#### 场景1：`n_splits=1`（你之前的用法）
- 抽样器只生成**1组**索引：比如“训练集9000条+测试集1000条”；
- 最终只得到1批1000条的评估样本；
- 适用：单次模型评估、快速对比不同模型（固定随机种子即可保证公平）。

#### 场景2：`n_splits=5`
- 抽样器生成**5组**独立的索引对：
  - 第1组：训练集9000条（索引0-8999）+ 测试集1000条（索引9000-9999）；
  - 第2组：训练集9000条（随机另选）+ 测试集1000条（随机另选）；
  - ... 共5组；
- 最终能得到5批独立的1000条评估样本；
- 适用：需要多次抽样验证模型稳定性（比如计算5次评估指标的平均值）。

#### 场景3：`n_splits=10`（默认值）
- 生成10组索引对，对应10批评估样本；
- 是sklearn默认的“10折分层抽样”，常用于模型的交叉验证（Stratified K-Fold）。

### 三、`n_splits`的使用场景（结合你的需求）
| `n_splits`值 | 适用场景 | 具体作用 |
|--------------|----------|----------|
| `1` | 单次模型评估/多模型对比 | 生成1批固定的评估样本，保证不同模型用同一批数据对比（你之前的核心需求）；逻辑简单，计算快 |
| `5~10` | 模型稳定性验证 | 生成多批独立的评估样本，计算指标的平均值/标准差（比如5次F1值的均值±标准差），避免单次抽样的偶然性；更严谨的正式评估 |
| `>10` | 大规模数据集的交叉验证 | 比如10万条数据做100折抽样，充分验证模型的泛化能力；但计算成本高，一般不用 |

### 四、关键注意事项（避坑）
1. **`n_splits`≠抽样数量**：
   - 抽样数量由`test_size`控制（比如`test_size=1000`表示每批抽1000条）；
   - `n_splits`只控制“生成多少批这样的样本”，而非单批样本数。

2. **随机种子的配合**：
   - 当`n_splits>1`时，`random_state`仍能保证“每次运行生成的多批样本完全一致”；
   - 比如`random_state=42`+`n_splits=5`，每次运行都会生成相同的5批评估样本，结果可复现。

3. **你的场景建议**：
   - 日常对比模型：保持`n_splits=1`即可，简单高效；
   - 正式上线前验证：改为`n_splits=5`，计算5次评估指标的平均值（比如剪枝蒸馏模型的F1值：0.6659±0.012），更能反映模型真实效果。
