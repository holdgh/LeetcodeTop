## 二、数据预处理（完整实操步骤，Windows适配）
### 1. 数据截取与划分（降低训练成本）
```python
import json
from sklearn.model_selection import train_test_split

# 1. 加载原始数据集（5万条）
with open("alpaca_data_zh_51k.json", "r", encoding="utf-8") as f:
    raw_data = json.load(f)

# 2. 截取1000条子集（适配笔记本）
sample_data = raw_data[:1000]

# 3. 按7:2:1划分训练集/验证集/测试集
train_data, rest_data = train_test_split(sample_data, test_size=0.3, random_state=42)
val_data, test_data = train_test_split(rest_data, test_size=1/3, random_state=42)

# 4. 保存为JSONL格式（datasets库适配）
def save_jsonl(data, path):
    with open(path, "w", encoding="utf-8") as f:
        for item in data:
            json.dump(item, f, ensure_ascii=False)
            f.write("\n")

save_jsonl(train_data, "train_data.jsonl")
save_jsonl(val_data, "val_data.jsonl")
save_jsonl(test_data, "test_data.jsonl")
```

### 2. 数据清洗（核心步骤，提升效果）
```python
import re
from difflib import SequenceMatcher

# 1. 去重函数（文本相似度≥90%判定为重复）
def is_similar(text1, text2, threshold=0.9):
    return SequenceMatcher(None, text1, text2).ratio() >= threshold

# 2. 去噪函数（删乱码、无效符号、超长文本）
def clean_text(text):
    # 删特殊符号/乱码
    text = re.sub(r"[^\u4e00-\u9fa5a-zA-Z0-9，。！？：；""''（）()、]", "", text)
    # 截断超长文本（指令微调单条文本≤512字符）
    return text[:512] if len(text) > 512 else text

# 3. 清洗训练集
cleaned_train = []
seen_texts = []
for item in train_data:
    # 清洗指令/输入/输出
    item["instruction"] = clean_text(item["instruction"])
    item["input"] = clean_text(item["input"])
    item["output"] = clean_text(item["output"])
    
    # 跳过空文本
    if not item["instruction"] or not item["output"]:
        continue
    
    # 去重
    combined_text = item["instruction"] + item["input"] + item["output"]
    is_dup = False
    for seen in seen_texts:
        if is_similar(combined_text, seen):
            is_dup = True
            break
    if not is_dup:
        seen_texts.append(combined_text)
        cleaned_train.append(item)

# 保存清洗后的数据
save_jsonl(cleaned_train, "train_data_cleaned.jsonl")
print(f"清洗后训练集样本数：{len(cleaned_train)}（原{len(train_data)}）")
```

### 3. 数据格式标准化（适配模型输入）
```python
from datasets import load_dataset
from transformers import AutoTokenizer

# 1. 加载Tokenizer
tokenizer = AutoTokenizer.from_pretrained("microsoft/phi-2", trust_remote_code=True)
tokenizer.pad_token = tokenizer.eos_token  # 生成模型必须设置pad_token 【注意这里并非所有的模型都会用到，有的模型的tokenizer已经具备pad_token】

# 2. 加载清洗后的数据
dataset = load_dataset("json", data_files={
    "train": "train_data_cleaned.jsonl",
    "validation": "val_data.jsonl"
})

# 3. 格式化函数（指令微调Prompt模板）
def format_function(examples):
    prompts = []
    labels = []
    for inst, inp, out in zip(examples["instruction"], examples["input"], examples["output"]):
        # 拼接Prompt（适配Phi-2的输入格式）
        prompt = f"### Instruction:\n{inst}\n### Input:\n{inp}\n### Response:\n"
        prompts.append(prompt)
        # 标签为输出文本（模型仅学习输出部分）
        labels.append(out)
    
    # 编码输入
    model_inputs = tokenizer(prompts, max_length=512, truncation=True, padding="max_length")
    # 编码标签
    label_inputs = tokenizer(labels, max_length=256, truncation=True, padding="max_length")
    model_inputs["labels"] = label_inputs["input_ids"]
    
    return model_inputs

# 4. 预处理数据集（批量处理，提升效率）
tokenized_dataset = dataset.map(
    format_function,
    batched=True,
    remove_columns=dataset["train"].column_names  # 删除原始列，仅保留编码后的数据
)
```

## 三、微调参数设置（适配Windows笔记本）
### 1. LoRA参数（核心，极简适配）
```python
from peft import LoraConfig

lora_config = LoraConfig(
    r=8,  # 低秩维度，适配小模型，避免过拟合
    lora_alpha=16,  # r的2倍，固定配比
    target_modules=["q_proj", "v_proj"],  # Phi-2核心注意力层
    lora_dropout=0.05,  # 防过拟合
    bias="none",
    task_type="CAUSAL_LM"  # 生成类任务
)
```

### 2. 训练参数（Windows硬件适配）
```python
from transformers import TrainingArguments

training_args = TrainingArguments(
    output_dir="./phi2-finetune-result",  # 训练结果保存路径
    per_device_train_batch_size=1,  # Windows CPU/低配N卡设为1
    gradient_accumulation_steps=4,  # 等效batch_size=4，省显存/CPU资源
    learning_rate=2e-4,  # LoRA专用学习率（Phi-2适配）
    num_train_epochs=3,  # 小模型3轮足够，避免过拟合
    logging_steps=5,  # 每5步打印日志，便于监控
    evaluation_strategy="epoch",  # 每轮评估验证集
    save_strategy="epoch",  # 每轮保存模型
    load_best_model_at_end=True,  # 训练结束加载最优模型
    fp16=False,  # CPU训练禁用，有N卡可设为True
    gradient_checkpointing=True,  # 省显存/内存（CPU训练也生效）
    weight_decay=0.01,  # 防过拟合
    warmup_ratio=0.1  # 学习率预热，稳训练
)
```

## 四、微调代码要点解析（完整可运行）
```python
from transformers import AutoModelForCausalLM, Trainer
from peft import get_peft_model
import swanlab  # 训练监控（可选，替代TensorBoard）

# ====================== 1. 初始化监控（可选） ======================
swanlab.init(project="phi2-zh-finetune", experiment_name="windows-laptop-test")

# ====================== 2. 加载模型 ======================
model = AutoModelForCausalLM.from_pretrained(
    "microsoft/phi-2",
    device_map="auto",  # 自动分配设备（CPU/N卡）
    trust_remote_code=True,
    torch_dtype="auto"  # 自动适配数据类型（CPU用float32，GPU用float16）
)

# ====================== 3. 注入LoRA适配器 ======================
model = get_peft_model(model, lora_config)
model.print_trainable_parameters()  # 打印可训练参数（约0.5%，极省资源）
# 输出示例：trainable params: 1,048,576 || all params: 2,730,035,200 || trainable%: 0.0384

# ====================== 4. 定义Trainer ======================
trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_dataset["train"],
    eval_dataset=tokenized_dataset["validation"]
)

# ====================== 5. 启动训练 ======================
trainer.train()

# ====================== 6. 保存模型 ======================
# 仅保存LoRA适配器（体积＜10MB，便于部署）
model.save_pretrained("./phi2-lora-adapter")
tokenizer.save_pretrained("./phi2-lora-adapter")

# ====================== 代码核心要点解析 ======================
# 1. device_map="auto"：Windows下自动识别CPU/GPU，无需手动指定；
# 2. get_peft_model：仅训练LoRA适配器参数，99.5%的模型参数冻结，省资源；
# 3. gradient_accumulation_steps=4：CPU训练时，用梯度累加等效增大batch_size，提升训练稳定性；
# 4. load_best_model_at_end=True：自动选择验证集效果最好的模型，避免过拟合；
# 5. 仅保存LoRA适配器：无需保存完整模型（2.7B），仅保存适配器（＜10MB），节省磁盘空间。
```

## 五、微调结果分析与优化（核心学习点）
### 1. 核心监控指标（SwanLab/日志）
| 指标                | 正常表现                          | 异常分析与优化                          |
|---------------------|-----------------------------------|-----------------------------------------|
| 训练集Loss          | 稳步下降（如从4.0→2.0）| 波动大：降低学习率/增大梯度累加步数      |

| 训练速度            | CPU：约0.5~1 step/s               | 速度过慢：进一步截取样本（如500条）|

### 2. 定性分析（人工校验）
- **采样测试**：从测试集随机选10条样本，用微调后模型推理，对比微调前后效果：
  ```python
  # 推理代码（Windows笔记本运行）
  from peft import PeftModel

  # 加载基础模型+LoRA适配器
  base_model = AutoModelForCausalLM.from_pretrained("microsoft/phi-2", device_map="auto")
  finetuned_model = PeftModel.from_pretrained(base_model, "./phi2-lora-adapter")

  # 测试案例
  def generate(inst, inp=""):
      prompt = f"### Instruction:\n{inst}\n### Input:\n{inp}\n### Response:\n"
      inputs = tokenizer(prompt, return_tensors="pt").to("cpu")  # 强制CPU（无N卡）
      outputs = finetuned_model.generate(
          **inputs,
          max_new_tokens=128,
          temperature=0.7,
          top_p=0.9,
          do_sample=True
      )
      return tokenizer.decode(outputs[0], skip_special_tokens=True).split("### Response:\n")[-1]

  # 测试1：创作类
  print(generate("写一段关于春天的中文小诗", ""))
  # 测试2：问答类
  print(generate("什么是大模型LoRA微调？", ""))
  ```
- **效果判定**：
  ✅ 微调后：回答更贴合中文语境、指令理解更精准、无乱码/重复；
  ❌ 微调前：回答可能偏英文语境、指令理解模糊。

### 3. 针对性优化方案（新手易落地）
| 问题现象                | 优化动作（Windows适配）|
|-------------------------|---------------------------------------------|
| 过拟合（验证集Loss上升） | 1. 减epochs至2轮 2. 增大weight_decay至0.05 3. 进一步清洗数据 |
| 欠拟合（效果无提升）| 1. 提学习率至3e-4 2. 增大LoRA的r值至16 3. 增加样本量至2000条 |
| 生成内容重复/乱码       | 1. 清洗训练集重复样本 2. 调低temperature至0.5 3. 截断超长输出 |

## 六、数据集对微调结果的分析与优化（核心学习点）
### 1. 数据集影响维度
| 数据维度       | 对结果的影响                          | 优化方向                                  |
|----------------|---------------------------------------|-------------------------------------------|
| 样本量         | 1000条：基础效果；500条：效果弱；2000条：效果提升 | 新手先练1000条，熟练后逐步增加            |
| 数据质量       | 脏数据→模型学错、答非所问              | 强化去重/去噪，人工抽检10%样本            |
| 数据场景匹配度 | 通用指令数据→仅适配通用场景            | 若做垂直场景（如客服），替换为客服指令数据 |
| 格式统一性     | 格式混乱→模型学不到规律                | 严格按`instruction+input+output`格式化    |

### 2. 数据集优化实操（进阶）
- **场景定制**：将通用Alpaca-zh替换为垂直场景数据（如「电商客服指令集」），对比微调效果，理解「数据对齐任务」的重要性；
- **样本均衡**：若训练集中「创作类」样本占比80%，补充「问答/总结」类样本，使各类别占比≤30%，避免模型偏科；
- **数据增强**：对小样本（＜500条）做同义词替换/句子重排，代码示例：
  ```python
  import synonyms  # 中文同义词库（需安装：pip install synonyms）

  def augment_text(text):
      # 同义词替换（保留核心语义）
      words = text.split()
      augmented_words = []
      for word in words:
          if len(word) > 1:
              syns = synonyms.nearby(word)[0]
              if syns:
                  augmented_words.append(syns[0])
              else:
                  augmented_words.append(word)
          else:
              augmented_words.append(word)
      return " ".join(augmented_words)

  # 对训练集output做增强
  for item in cleaned_train:
      item["output"] = augment_text(item["output"])
  ```
