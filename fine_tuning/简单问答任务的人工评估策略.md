## 简单问答任务人工评估清单
### 评估维度（共5项，每项0-2分，总分0-10分）
| 评估维度 | 评分标准（0-2分） | 示例说明 |
|----------|-------------------|----------|
| **1. 回答相关性** | 2分：回答完全匹配问题核心需求<br>1分：回答部分匹配，但有无关信息<br>0分：回答与问题无关 | 问题：“苹果的英文是什么？”<br>2分：“苹果的英文是apple”<br>0分：“香蕉的英文是banana” |
| **2. 回答准确性** | 2分：内容完全正确，无错误<br>1分：内容基本正确，但有轻微错误（如用词偏差）<br>0分：内容错误或存在误导性信息 | 问题：“中国首都是哪里？”<br>2分：“中国的首都是北京”<br>1分：“中国的首都是北平（注：北平是北京旧称，未明确说明）” |
| **3. 回答简洁性** | 2分：语言简洁，无冗余信息<br>1分：有少量冗余，但核心信息明确<br>0分：内容冗长、逻辑混乱 | 问题：“1+1等于几？”<br>2分：“1+1等于2”<br>0分：“在数学的基础运算中，1加1的结果通常是2，不过在二进制中可能有不同情况……”（问题未提及特殊场景） |
| **4. 语言流畅性** | 2分：语句通顺，符合中文表达习惯<br>1分：语句基本通顺，但有轻微语病<br>0分：语句不通顺，无法理解 | 问题：“今天天气怎么样？”<br>2分：“今天天气晴朗，气温25度”<br>1分：“今天的天气是晴朗，25度的气温” |
| **5. 格式规范性** | 2分：格式清晰（如问答对应、无乱码）<br>1分：格式基本规范，但有轻微排版问题<br>0分：格式混乱（如包含特殊符号、乱码） | 问题：“请列出3种水果”<br>2分：“常见的3种水果有：苹果、香蕉、橙子”<br>0分：“水果有<<|im_end|>苹果香蕉橙子” |


### 评估操作步骤
1. **抽样**：从验证集中随机抽取10-20条样本（覆盖不同问题类型）；
2. **评分**：按上述维度对每条样本打分，计算单条样本的总分；
3. **统计**：计算所有样本的平均分（≥8分表示效果优秀，6-8分良好，＜6分需优化）。

import json
import numpy as np
import pandas as pd
import nltk
from nltk.translate.bleu_score import sentence_bleu, SmoothingFunction
from rouge_score import rouge_scorer
import matplotlib.pyplot as plt

# 下载nltk必要资源（首次运行需执行）
nltk.download('punkt')

# ====================== 核心评估函数 ======================
class QA_Evaluator:
    def __init__(self):
        # 初始化BLEU平滑函数（解决短文本评分偏低问题）
        self.smoother = SmoothingFunction().method4
        # 初始化ROUGE评估器（仅关注ROUGE-L，适配问答任务的完整匹配）
        self.rouge_scorer = rouge_scorer.RougeScorer(['rougeL'], use_stemmer=False)

    def calculate_bleu(self, reference: str, candidate: str) -> float:
        """
        计算单条样本的BLEU分数（适配短文本问答）
        :param reference: 标准答案（真实回答）
        :param candidate: 模型生成回答
        :return: BLEU分数（0-1）
        """
        # 分词（中文按字符拆分，英文按单词）
        ref_tokens = list(reference) if self.is_chinese(reference) else nltk.word_tokenize(reference.lower())
        cand_tokens = list(candidate) if self.is_chinese(candidate) else nltk.word_tokenize(candidate.lower())
        
        # 计算BLEU（权重：1-gram占100%，适配短文本）
        bleu = sentence_bleu(
            [ref_tokens], 
            cand_tokens,
            weights=(1.0, 0.0, 0.0, 0.0),  # 仅用1-gram，避免短文本分数为0
            smoothing_function=self.smoother
        )
        return round(bleu, 4)

    def calculate_rouge_l(self, reference: str, candidate: str) -> dict:
        """
        计算单条样本的ROUGE-L分数
        :param reference: 标准答案
        :param candidate: 模型生成回答
        :return: 包含precision/recall/fmeasure的字典
        """
        scores = self.rouge_scorer.score(reference, candidate)
        rouge_l = {
            "precision": round(scores['rougeL'].precision, 4),
            "recall": round(scores['rougeL'].recall, 4),
            "fmeasure": round(scores['rougeL'].fmeasure, 4)
        }
        return rouge_l

    def evaluate_batch(self, data: list) -> pd.DataFrame:
        """
        批量评估样本，返回详细评分表
        :param data: 样本列表，每个样本格式：{"question": "", "reference": "", "candidate": ""}
        :return: 包含所有指标的DataFrame
        """
        results = []
        for idx, sample in enumerate(data):
            question = sample["question"]
            reference = sample["reference"]
            candidate = sample["candidate"]
            
            # 计算指标
            bleu = self.calculate_bleu(reference, candidate)
            rouge_l = self.calculate_rouge_l(reference, candidate)
            
            results.append({
                "sample_id": idx + 1,
                "question": question,
                "reference": reference,
                "candidate": candidate,
                "bleu": bleu,
                "rouge_l_precision": rouge_l["precision"],
                "rouge_l_recall": rouge_l["recall"],
                "rouge_l_f1": rouge_l["fmeasure"]
            })
        
        # 转换为DataFrame并计算整体均值
        df = pd.DataFrame(results)
        avg_metrics = {
            "avg_bleu": round(df["bleu"].mean(), 4),
            "avg_rouge_l_precision": round(df["rouge_l_precision"].mean(), 4),
            "avg_rouge_l_recall": round(df["rouge_l_recall"].mean(), 4),
            "avg_rouge_l_f1": round(df["rouge_l_f1"].mean(), 4)
        }
        return df, avg_metrics

    def is_chinese(self, text: str) -> bool:
        """判断文本是否为中文"""
        for char in text:
            if '\u4e00' <= char <= '\u9fff':
                return True
        return False

    def plot_metrics(self, metrics_05B: dict, metrics_7B: dict, save_path: str = "eval_metrics.png"):
        """
        可视化0.5B和7B模型的指标对比
        :param metrics_05B: 0.5B模型的平均指标
        :param metrics_7B: 7B模型的平均指标
        :param save_path: 图片保存路径
        """
        # 准备数据
        metrics = ["BLEU", "ROUGE-L Precision", "ROUGE-L Recall", "ROUGE-L F1"]
        values_05B = [
            metrics_05B["avg_bleu"],
            metrics_05B["avg_rouge_l_precision"],
            metrics_05B["avg_rouge_l_recall"],
            metrics_05B["avg_rouge_l_f1"]
        ]
        values_7B = [
            metrics_7B["avg_bleu"],
            metrics_7B["avg_rouge_l_precision"],
            metrics_7B["avg_rouge_l_recall"],
            metrics_7B["avg_rouge_l_f1"]
        ]
        
        # 绘图
        x = np.arange(len(metrics))
        width = 0.35
        
        plt.rcParams["font.sans-serif"] = ["SimHei"]  # 显示中文
        plt.figure(figsize=(12, 6))
        plt.bar(x - width/2, values_05B, width, label="千问0.5B")
        plt.bar(x + width/2, values_7B, width, label="千问7B")
        
        # 添加数值标签
        for i, v in enumerate(values_05B):
            plt.text(i - width/2, v + 0.01, str(v), ha="center")
        for i, v in enumerate(values_7B):
            plt.text(i + width/2, v + 0.01, str(v), ha="center")
        
        plt.xlabel("评估指标")
        plt.ylabel("分数（0-1）")
        plt.title("千问0.5B vs 7B 简单问答任务自动评估结果")
        plt.xticks(x, metrics)
        plt.ylim(0, 1.0)
        plt.legend()
        plt.grid(axis="y", linestyle="--", alpha=0.7)
        plt.savefig(save_path, bbox_inches="tight", dpi=300)
        plt.close()
        print(f"✅ 指标对比图已保存至：{save_path}")

# ====================== 示例：加载数据并评估 ======================
if __name__ == "__main__":
    # 1. 初始化评估器
    evaluator = QA_Evaluator()
    
    # 2. 准备测试数据（替换为你的验证集数据）
    # 格式：[{"question": 问题, "reference": 标准答案, "candidate": 模型回答}]
    test_data_05B = [

            "reference": "《静夜思》的作者是李白",
            "candidate": "李白是《静夜思》的作者"  # 0.5B模型回答
        },
        {

        },
        {

            "reference": "苹果的英文是apple",
            "candidate": "苹果的英文是appl"  # 0.5B模型回答（轻微错误）
        }
    ]
    
    test_data_7B = [
        {
            "question": "《静夜思》的作者是谁？",
            "reference": "《静夜思》的作者是李白",
            "candidate": "《静夜思》的作者是唐代诗人李白"  # 7B模型回答
        },
        {
            "question": "1+1等于几？",
            "reference": "1+1等于2",
            "candidate": "1+1等于2"  # 7B模型回答
        },
        {
            "question": "苹果的英文是什么？",
            "reference": "苹果的英文是apple",
            "candidate": "苹果的英文是apple"  # 7B模型回答
        }
    ]
    
    # 3. 评估0.5B模型
    df_05B, avg_05B = evaluator.evaluate_batch(test_data_05B)
    print("===== 千问0.5B模型评估结果 =====")
    print("详细评分表：")
    print(df_05B)
    print("\n平均指标：")
    print(avg_05B)
    
    # 4. 评估7B模型
    df_7B, avg_7B = evaluator.evaluate_batch(test_data_7B)
    print("\n===== 千问7B模型评估结果 =====")
    print("详细评分表：")
    print(df_7B)
    print("\n平均指标：")
    print(avg_7B)
    
    # 5. 可视化对比
    evaluator.plot_metrics(avg_05B, avg_7B)
    
    # 6. 保存评估结果到CSV
    df_05B.to_csv("qwen_05B_eval_result.csv", index=False, encoding="utf-8-sig")
    df_7B.to_csv("qwen_7B_eval_result.csv", index=False, encoding="utf-8-sig")
    print("\n✅ 评估结果已保存为CSV文件")

### 3. 代码核心适配点（针对简单问答任务）
| 优化点 | 作用 |
|--------|------|
| BLEU仅用1-gram | 短文本问答的回答通常只有几个词/字符，多gram会导致分数为0，1-gram更贴合实际 |
| 中文按字符分词 | 解决中文无天然分词边界的问题，保证BLEU/ROUGE计算准确 |
| 平滑函数 | 避免因模型回答与标准答案略有差异导致BLEU分数为0 |
| 仅关注ROUGE-L | ROUGE-L衡量最长公共子序列，更适合问答任务的“完整匹配”需求（比ROUGE-1/2更有参考价值） |

### 4. 使用说明
1. **替换测试数据**：将 `test_data_05B`/`test_data_7B` 替换为你的验证集真实数据（从alpaca-zh-51k的验证集中抽取）；
2. **指标解读**：
   - BLEU：衡量回答的“匹配度”，分数越高越接近标准答案；
   - ROUGE-L F1：综合Precision（精准度）和Recall（召回率），是核心参考指标；
3. **结果对比**：通过可视化图可直观看到7B模型的指标是否优于0.5B（与你的Loss趋势一致）。

### 5. 注意事项
- 自动评估指标仅作为**辅助参考**，最终需结合人工评估（比如模型回答有小语法错误但核心正确，自动指标可能偏低，但人工评估仍可打高分）；
- 若问答任务包含专业术语，可在分词阶段加入专业词表，提升指标准确性。
